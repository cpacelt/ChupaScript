# ChupaScript — синтаксическая грамматика
#
# Нормативный артефакт. Определяет, какие последовательности токенов образуют
# единицу исполнения. Токены определены в grammar/lexical.txt.
#
# Нотация ECMA-262:
#   Имя :                        продукция синтаксической грамматики
#   Имя_opt                      необязательный элемент
#   one of a b c                 сокращение для списка альтернатив
#   [lookahead != x]             следующий токен не x
#   [lookahead !<- {a, b}]       следующий токен не принадлежит множеству
#   [no LineTerminator here]     между соседними элементами нет перевода строки
#   # текст                      комментарий, ненормативен
#
# Идентификатор правила = имя продукции. Тесты ссылаются на него.
# Ссылка в спецификации: `grammar/syntax.txt`.
#
# Грамматика допускает больше, чем допускает язык. Ограничения, для которых
# альтернативой является не другое прочтение текста, а отвергнутое прочтение,
# заданы статическими правилами (`05-names.md`), а не составом продукций. Так заданы:
# позиция стрелочной функции, допустимость вызываемого имени, корректность цели
# присваивания.


# =============================================================================
# 1. Цели разбора
# =============================================================================

# Целей две. Цель задаётся снаружи вместе с единицей исполнения (раздел 5.1) и из
# её текста не выводится.
#
# Разделение необходимо, потому что один и тот же текст читается по-разному в
# зависимости от цели. `{ }` по цели Script — пустой Block, по цели Expression —
# пустой объект. Выбрать прочтение статическим правилом нельзя: оно умеет
# отвергнуть разбор, но не перечитать текст заново.
#
# Expression служит и целью разбора, и нетерминалом внутри других продукций.
# Противоречия здесь нет: цель разбора — параметр разбора, а не свойство
# продукции. Отдельная продукция-обёртка с единственной альтернативой была бы
# сущностью без содержания.

Script :
    StatementList_opt

# Expression определён в разделе 2 вместе с прочими выражениями.

# Соответствие целям ECMA-262 (CS-CONF-0001): Script — Script,
# Expression — AssignmentExpression. Имя Expression обозначает здесь и в
# ECMA-262 разное: там эта продукция выводит оператор запятая, которого в языке
# нет.


# =============================================================================
# 2. Выражения
# =============================================================================

# Уровни приоритета следуют ECMA-262 с изъятием отсутствующих: оператора
# запятая, побитовых и сдвигов. Поэтому Expression совпадает с
# AssignmentExpression, а операндом && служит EqualityExpression.

# Expression — одна из двух целей разбора (раздел 1) и одновременно нетерминал
# в позициях, где допустимо одно выражение: скобки, условие if, обходимое
# выражение for...of, подстановка в шаблон, индекс при обращении к свойству.

Expression :
    AssignmentExpression


# --- 2.1. Первичные выражения -----------------------------------------------

PrimaryExpression :
    Identifier
    Literal
    ArrayLiteral
    ObjectLiteral
    TemplateLiteral
    ( Expression )

Literal :
    null
    true
    false
    NumericLiteral
    StringLiteral

# undefined, NaN и Infinity — Identifier, а не Literal: это имена,
# разрешаемые в предопределённые значения (lexical.txt, раздел 5).


# --- 2.2. Литерал массива ---------------------------------------------------

ArrayLiteral :
    [ ]
    [ ElementList ]

ElementList :
    AssignmentExpression
    ElementList , AssignmentExpression

# Пропуски (`[1, , 3]`) не допускаются: ElementList требует выражение между
# запятыми. Завершающая запятая не допускается.


# --- 2.3. Литерал объекта ---------------------------------------------------

ObjectLiteral :
    { }
    { PropertyDefinitionList }

PropertyDefinitionList :
    PropertyDefinition
    PropertyDefinitionList , PropertyDefinition

PropertyDefinition :
    Identifier
    PropertyName : AssignmentExpression

PropertyName :
    IdentifierName
    StringLiteral
    NumericLiteral
    [ AssignmentExpression ]

# Первая альтернатива PropertyDefinition — сокращённая запись `{ a }`.
# Ключом служит любой IdentifierName, включая зарезервированные слова:
# `{ default: 1 }` допустимо. Методов, get/set и распространения нет.


# --- 2.4. Шаблонные строки --------------------------------------------------

TemplateLiteral :
    NoSubstitutionTemplate
    SubstitutionTemplate

SubstitutionTemplate :
    TemplateHead Expression TemplateSpans

TemplateSpans :
    TemplateTail
    TemplateMiddleList TemplateTail

TemplateMiddleList :
    TemplateMiddle Expression
    TemplateMiddleList TemplateMiddle Expression


# --- 2.5. Обращение к свойству и вызов --------------------------------------

LeftHandSideExpression :
    MemberExpression
    CallExpression
    OptionalExpression

MemberExpression :
    PrimaryExpression
    MemberExpression . IdentifierName
    MemberExpression [ Expression ]

CallExpression :
    MemberExpression Arguments
    CallExpression Arguments
    CallExpression . IdentifierName
    CallExpression [ Expression ]

Arguments :
    ( )
    ( ArgumentList )

ArgumentList :
    AssignmentExpression
    ArgumentList , AssignmentExpression

# Грамматика допускает произвольное выражение в позиции вызываемого, включая
# `f(x)(y)` и `arr[0](x)`. Отвергаются они статическим правилом: множество
# функций фиксировано на момент проверки, поэтому «обозначает ли имя функцию»
# — вопрос, на который отвечает проверка, а не грамматика. Задавать часть
# ограничения продукциями означало бы делить одно понятие между двумя
# механизмами и давать одной ошибке две несравнимые диагностики.


# --- 2.6. Необязательные цепочки --------------------------------------------

OptionalExpression :
    MemberExpression OptionalChain
    CallExpression OptionalChain
    OptionalExpression OptionalChain

OptionalChain :
    ?. IdentifierName
    ?. [ Expression ]
    OptionalChain . IdentifierName
    OptionalChain [ Expression ]
    OptionalChain Arguments

# Звено `?. Arguments` отсутствует: `a?.()` — вызов значения, а функции
# значениями не являются. `a?.b()` допустимо — это вызов метода внутри цепочки,
# и при пустом `a` он не происходит.


# --- 2.7. Унарные операторы -------------------------------------------------

UnaryExpression :
    LeftHandSideExpression
    - UnaryExpression
    ! UnaryExpression
    typeof UnaryExpression

# delete, void и побитовое ~ отсутствуют.
#
# Унарного + тоже нет. Его единственное действие — ToNumber, а его уже
# выполняет Number(x) (раздел 9); второй записи одного и того же действия язык
# не сохраняет — на том же основании, на котором не сохранено возведение в
# степень при наличии Math.pow. Бинарный + в AdditiveExpression не затронут, и
# токен + остаётся в OtherPunctuator.
# Унарный - сохранён, а не изъят тем же способом: NumericLiteral знака не
# несёт, и без унарного минуса отрицательные числа стали бы невыразимы.
#
# Операторов инкремента и декремента нет ни в префиксной, ни в постфиксной
# форме. Увеличение записывается присваиванием: `i = i + 1`. Вместе с ними
# исчезает и отдельный уровень грамматики ECMA-262 между LeftHandSideExpression
# и UnaryExpression: он существовал только ради этих двух операторов.


# --- 2.8. Бинарные операторы ------------------------------------------------

MultiplicativeExpression :
    UnaryExpression
    MultiplicativeExpression MultiplicativeOperator UnaryExpression

MultiplicativeOperator : one of
    * / %

# Оператора возведения в степень нет; оно выражается вызовом Math.pow
# (раздел 9). Вместе с ним отпало и ограничение JavaScript, по которому унарный
# минус слева от этого оператора — ошибка разбора: без оператора запрещать
# нечего.

AdditiveExpression :
    MultiplicativeExpression
    AdditiveExpression + MultiplicativeExpression
    AdditiveExpression - MultiplicativeExpression

RelationalExpression :
    AdditiveExpression
    RelationalExpression < AdditiveExpression
    RelationalExpression > AdditiveExpression
    RelationalExpression <= AdditiveExpression
    RelationalExpression >= AdditiveExpression

EqualityExpression :
    RelationalExpression
    EqualityExpression == RelationalExpression
    EqualityExpression != RelationalExpression
    EqualityExpression === RelationalExpression
    EqualityExpression !== RelationalExpression

# Операторы in и instanceof отсутствуют. Наличие свойства проверяется вызовом
# Object.hasOwn (раздел 9).
#
# Оператор in в JavaScript учитывает и унаследованные имена, а цепочки
# прототипов в языке нет (приложение C, CS-DIV-0001). Сохранив in, пришлось бы
# либо описывать вторую операцию с тем же смыслом, что у Object.hasOwn, либо
# определять её через механизм, которого в языке не существует.
#
# Параметр [In] грамматике не нужен: в ECMA-262 он существует ради оператора in
# в заголовке `for ( ; ; )`, здесь нет ни того, ни другого.


# --- 2.9. Логические операторы ----------------------------------------------

LogicalANDExpression :
    EqualityExpression
    LogicalANDExpression && EqualityExpression

LogicalORExpression :
    LogicalANDExpression
    LogicalORExpression || LogicalANDExpression

CoalesceExpression :
    CoalesceExpressionHead ?? EqualityExpression

CoalesceExpressionHead :
    CoalesceExpression
    EqualityExpression

ShortCircuitExpression :
    LogicalORExpression
    CoalesceExpression

# Форма CoalesceExpression запрещает смешивать ?? с && и || без скобок:
# `a ?? b || c` и `a && b ?? c` — ошибки разбора, требуется `(a ?? b) || c`.
# Так же в JavaScript.


# --- 2.10. Условный оператор и присваивание ---------------------------------

ConditionalExpression :
    ShortCircuitExpression
    ShortCircuitExpression ? AssignmentExpression : AssignmentExpression

AssignmentExpression :
    ConditionalExpression
    ArrowFunction
    LeftHandSideExpression = AssignmentExpression
    LeftHandSideExpression AssignmentOperator AssignmentExpression

AssignmentOperator : one of
    += -= *= /= %=

# Составных присваиваний с возведением в степень и с побитовыми операторами
# нет, потому что нет самих операторов.
#
# Грамматика допускает любое LeftHandSideExpression слева от знака
# присваивания, включая `1 = x`. Пригодность цели проверяется статическим
# правилом; так же устроена ECMA-262.
#
# ArrowFunction входит сюда наравне с прочими альтернативами. Требование
# «только непосредственно аргументом вызова» — статическое правило: вторая его
# половина, «вызываемое принимает функцию», грамматике недоступна, а делить
# одно требование надвое незачем.


# =============================================================================
# 3. Statement и объявления
# =============================================================================

StatementList :
    StatementListItem
    StatementList StatementListItem

StatementListItem :
    Statement
    Declaration

Declaration :
    LexicalDeclaration
    FunctionDeclaration

Statement :
    Block
    ExpressionStatement
    IfStatement
    ForOfStatement
    BreakStatement
    ContinueStatement
    ReturnStatement

# Пустой Statement (одиночная `;`) отсутствует.
#
# Меток нет, поэтому продукция LabelledStatement отсутствует, а `a: 1` в
# позиции Statement — ошибка разбора при любой цели.
#
# При цели Expression списка StatementListItem нет вовсе: она выводит
# выражение.


# --- 3.1. Блок ---------------------------------------------------------------

Block :
    { StatementList_opt }


# --- 3.2. Объявление переменной ---------------------------------------------

LexicalDeclaration :
    LetOrConst LexicalBinding ;

LetOrConst : one of
    let const

LexicalBinding :
    Identifier Initializer
    BindingPattern Initializer

Initializer :
    = AssignmentExpression

# Один связыватель на объявление: `let a = 1, b = 2;` не допускается.
#
# Инициализатор обязателен и при let: `let x;` не допускается. Значение
# объявленного имени тем самым задано в точке объявления всегда.


# --- 3.3. Образцы ------------------------------------------------------------

BindingPattern :
    ObjectPattern
    ArrayPattern

ObjectPattern :
    { }
    { BindingPropertyList }

BindingPropertyList :
    BindingProperty
    BindingPropertyList , BindingProperty

BindingProperty :
    Identifier
    PropertyName : Identifier

ArrayPattern :
    [ ]
    [ BindingElementList ]

BindingElementList :
    Identifier
    BindingElementList , Identifier

# Образцы плоские: целью связывания всегда служит Identifier, поэтому
# `let { a: { b } } = obj;` не допускается. Значений по умолчанию и остатка
# (`...rest`) нет. Пропуски в ArrayPattern не допускаются.


# --- 3.4. Выражение как Statement -------------------------------------------

ExpressionStatement :
    [lookahead !<- {{, function}] Expression ;

# Ограничение отделяет Block от литерала объекта и FunctionDeclaration от
# выражения. Поэтому `{ a: 1 }` в позиции Statement читается как Block и
# отвергается; значением его делает запись `({ a: 1 })`. При цели
# Expression ограничение неприменимо, и `{ a: 1 }` там — объект.


# --- 3.5. Условный Statement ------------------------------------------------

IfStatement :
    if ( Expression ) Block
    if ( Expression ) Block else Block
    if ( Expression ) Block else IfStatement

# Ветви — только Block. Запись `if (x) doIt();` не допускается. Это сужение
# относительно JavaScript устраняет неоднозначность «висящего else»: `else`
# всегда относится к единственному if, который может его принять, и грамматика
# однозначна без дополнительного правила.


# --- 3.6. Цикл ---------------------------------------------------------------

ForOfStatement :
    for ( LetOrConst ForBinding of Expression ) Block

ForBinding :
    Identifier
    BindingPattern

# Единственная форма цикла. `while`, `do...while`, `for ( ; ; )` и `for...in`
# отсутствуют.
#
# Переменная цикла объявляется обязательно: формы `for (x of items)` с уже
# существующим именем нет.
#
# Образец в заголовке допускается тот же, что в объявлении:
# `for (let [key, value] of Object.entries(o))`.

BreakStatement :
    break ;

ContinueStatement :
    continue ;

# Меток нет, поэтому `break` и `continue` относятся к ближайшему объемлющему
# ForOfStatement. Отсутствие такового — статическая ошибка.


# =============================================================================
# 4. Функции
# =============================================================================

FunctionDeclaration :
    function Identifier ( FormalParameterList_opt ) { FunctionBody }

FormalParameterList :
    FormalParameter
    FormalParameterList , FormalParameter

FormalParameter :
    Identifier
    Identifier = Literal

FunctionBody :
    StatementList_opt

ReturnStatement :
    return ;
    return [no LineTerminator here] Expression ;

# Функциональных выражений нет: function в позиции выражения недостижим.
# Параметр-остаток (`...rest`) отсутствует, образцы в параметрах отсутствуют.
#
# Значением по умолчанию служит Literal, а не произвольное выражение. При
# произвольном выражении список параметров стал бы областью: инициализатор в
# JavaScript видит предыдущие параметры (`function f(a, b = a + 1)`), а
# обращение к последующим даёт ошибку. Литерал этот вопрос упраздняет —
# вычислять нечего, значение известно на разборе. Ссылка на другой параметр
# записывается в теле функции.

ArrowFunction :
    ArrowParameters [no LineTerminator here] => ConciseBody

ArrowParameters :
    Identifier
    ( )
    ( ArrowParameterList )

ArrowParameterList :
    Identifier
    ArrowParameterList , Identifier

ConciseBody :
    [lookahead != {] Expression
    { FunctionBody }

# Параметрами служат только Identifier: ни образцов, ни значений по умолчанию.
#
# При иных параметрах один и тот же текст выводился бы двумя способами, а
# различающий его токен стоит за парной `)`: `({ a })` — литерал объекта либо
# образец, `(a = 1)` — скобочное присваивание либо параметр со значением по
# умолчанию. ECMA-262 выводит такой текст накрывающей продукцией
# (CoverParenthesizedExpressionAndArrowParameterList) и уточняет вывод, узнав,
# следует ли за скобками `=>`. Уточнение опирается на нормативное отношение
# «какое выражение обратимо в связывание», которое в языке больше нигде не
# требуется: после let и const и в заголовке for...of BindingPattern выводится
# грамматикой напрямую.
#
# Образцы и значения по умолчанию требуют этого отношения в равной мере,
# промежуточного варианта нет: либо параметры — Identifier и отношения не
# существует, либо параметры — полные связыватели вместе с обеими
# конструкциями. Принят первый вариант: обращение к свойству в теле
# (`x => x.id`) заменяет образец без потери выразительности.
#
# Пересечения на `()` и `(a, b)` не возникает: оператора запятая нет, поэтому
# скобочное выражение содержит ровно одно выражение.
#
# Ограничение [lookahead != {] в ConciseBody отделяет тело-блок от литерала
# объекта, как в ECMA-262: `x => ({ id: x })` возвращает объект, а в
# `x => { id: x }` фигурная скобка начинает тело. В JavaScript такое тело
# содержит метку `id` и вычисляет undefined; здесь меток нет, поэтому это
# ошибка разбора — сужение, инвариант CS-CONF-0001 сохраняется.


# =============================================================================
# 5. Ограничения на перевод строки
# =============================================================================

# Автоматической расстановки точек с запятой в языке нет (CS-LEX-0004). Тем не
# менее ограничения [no LineTerminator here] сохранены во всех местах, где они
# есть в ECMA-262 и где соответствующая конструкция существует здесь. Причина в
# инварианте раздела 1: в этих двух местах перевод строки меняет прочтение
# текста JavaScript. Без ограничений единица исполнения либо получила бы смысл,
# отличный от своего же смысла в JavaScript (нарушение CS-CONF-0010), либо была
# бы принята там, где JavaScript отвергает (нарушение CS-CONF-0001).

# | Продукция          | Между чем         | Что было бы иначе                    |
# |--------------------|-------------------|--------------------------------------|
# | ReturnStatement    | return и Expression | `return \n x;` в JavaScript —      |
# |                    |                   | `return; x;`, здесь стало бы         |
# |                    |                   | возвратом значения x                 |
# | ArrowFunction      | параметры и =>    | `x \n => x` JavaScript отвергает,    |
# |                    |                   | здесь было бы принято                |

# Наличие перевода строки между токенами определено правилом CS-LEX-0003: оно
# учитывает и переводы строк внутри многострочного комментария.
